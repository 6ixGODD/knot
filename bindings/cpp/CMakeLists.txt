cmake_minimum_required(VERSION 3.23)
project(knot_cpp_bridges LANGUAGES CXX)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
get_filename_component(KNOT_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_private.py"
    "${KNOT_SOURCE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
foreach(variant IN ITEMS shared static)
    add_library(knot_cpp_${variant} STATIC src/codec.cpp)
    target_compile_features(knot_cpp_${variant} PUBLIC cxx_std_17)
    target_include_directories(knot_cpp_${variant} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${KNOT_SOURCE_ROOT}/contracts/abi")
    target_include_directories(knot_cpp_${variant} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${KNOT_SOURCE_ROOT}/vendor/nlohmann")
    set_target_properties(knot_cpp_${variant} PROPERTIES UNITY_BUILD OFF DISABLE_PRECOMPILE_HEADERS ON
        POSITION_INDEPENDENT_CODE ON MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    if(MSVC)
        target_compile_options(knot_cpp_${variant} PRIVATE /W4 /WX /EHsc /utf-8)
    else()
        target_compile_options(knot_cpp_${variant} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
target_compile_definitions(knot_cpp_static PUBLIC KNOT_STATIC)
