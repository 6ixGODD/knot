cmake_minimum_required(VERSION 3.20)
project(knot_json_coexistence LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT EXISTS "${HOST_JSON_DIR}/json.hpp")
  message(FATAL_ERROR "HOST_JSON_DIR must contain the pinned 3.11.3 test header")
endif()
get_filename_component(KNOT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(KNOT_JSON_DIR "${KNOT_ROOT}/vendor/nlohmann")
file(SHA256 "${KNOT_JSON_DIR}/json.hpp" KNOT_JSON_SHA)
file(SHA256 "${HOST_JSON_DIR}/json.hpp" HOST_JSON_SHA)
if(NOT KNOT_JSON_SHA STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
  message(FATAL_ERROR "Bundled candidate header hash mismatch")
endif()
if(NOT HOST_JSON_SHA STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
  message(FATAL_ERROR "Host fixture header hash mismatch")
endif()
message(STATUS "C++ compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
message(STATUS "System: ${CMAKE_SYSTEM_NAME} ${CMAKE_SYSTEM_VERSION} ${CMAKE_SYSTEM_PROCESSOR}")
configure_file(headers.hpp.in headers.hpp @ONLY)
foreach(probe IN ITEMS host_first bundled_first host_diagnostics_first)
  add_executable(${probe} ${probe}.cpp)
  target_include_directories(${probe} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  add_test(NAME ${probe} COMMAND ${probe})
endforeach()
add_executable(separate_translation_units separate_translation_units.cpp host_translation_unit.cpp bundled_translation_unit.cpp)
target_include_directories(separate_translation_units PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME separate_translation_units COMMAND separate_translation_units)
add_executable(abi_c abi.c)
target_include_directories(abi_c PRIVATE "${KNOT_ROOT}/contracts/abi")
target_compile_definitions(abi_c PRIVATE KNOT_STATIC)
add_test(NAME abi_c COMMAND abi_c)
add_executable(abi_cpp abi_cpp.cpp)
target_include_directories(abi_cpp PRIVATE "${KNOT_ROOT}/contracts/abi")
target_compile_definitions(abi_cpp PRIVATE KNOT_STATIC)
add_test(NAME abi_cpp COMMAND abi_cpp)
