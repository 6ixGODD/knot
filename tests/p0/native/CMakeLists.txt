cmake_minimum_required(VERSION 3.20)
project(knot_native_p0 LANGUAGES C CXX)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
find_package(Threads REQUIRED)
add_executable(abi_c ../json-coexistence/abi.c)
add_executable(abi_cpp ../json-coexistence/abi_cpp.cpp)
foreach(TARGET IN ITEMS abi_c abi_cpp)
    target_include_directories(${TARGET} PRIVATE ../../../contracts/abi)
    add_test(NAME ${TARGET} COMMAND ${TARGET})
endforeach()
add_library(probe_shared SHARED IMPORTED)
set_target_properties(probe_shared PROPERTIES IMPORTED_LOCATION "${P0_SHARED}")
if(WIN32)
    set_target_properties(probe_shared PROPERTIES IMPORTED_IMPLIB "${P0_IMPLIB}")
endif()
add_library(probe_static STATIC IMPORTED)
set_target_properties(probe_static PROPERTIES IMPORTED_LOCATION "${P0_STATIC}")
target_compile_definitions(probe_static INTERFACE P0_STATIC)
set(P0_FRAMEWORK_FLAG "")
foreach(ITEM IN LISTS P0_SYSTEM_LIBS)
    if(P0_FRAMEWORK_FLAG)
        # Keep each framework and its name in one CMake link item. Otherwise
        # the bare name becomes -l<Name> and repeated -framework flags collapse.
        target_link_libraries(probe_static INTERFACE "${P0_FRAMEWORK_FLAG} ${ITEM}")
        set(P0_FRAMEWORK_FLAG "")
    elseif(ITEM STREQUAL "-framework")
        set(P0_FRAMEWORK_FLAG "${ITEM}")
    elseif(MSVC AND ITEM MATCHES "^/")
        target_link_options(probe_static INTERFACE "${ITEM}")
    else()
        target_link_libraries(probe_static INTERFACE "${ITEM}")
    endif()
endforeach()
if(P0_FRAMEWORK_FLAG)
    message(FATAL_ERROR "Missing framework name in P0_SYSTEM_LIBS")
endif()
add_subdirectory(../json-isolation json-isolation)
foreach(MODE IN ITEMS shared static)
    add_library(cpp_bridge_${MODE} STATIC bridge.cpp)
    target_link_libraries(cpp_bridge_${MODE} PRIVATE private_codec PUBLIC probe_${MODE})
    set_target_properties(cpp_bridge_${MODE} PROPERTIES UNITY_BUILD OFF DISABLE_PRECOMPILE_HEADERS ON)
    add_executable(native_c_${MODE} consumer.c)
    target_link_libraries(native_c_${MODE} PRIVATE probe_${MODE})
    add_executable(native_cpp_${MODE} consumer.cpp)
    target_link_libraries(native_cpp_${MODE} PRIVATE cpp_bridge_${MODE} Threads::Threads)
    foreach(LANGUAGE IN ITEMS c cpp)
        add_test(NAME native_${LANGUAGE}_${MODE} COMMAND native_${LANGUAGE}_${MODE})
        if(WIN32 AND MODE STREQUAL "shared")
            add_custom_command(TARGET native_${LANGUAGE}_${MODE} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "${P0_SHARED}" "$<TARGET_FILE_DIR:native_${LANGUAGE}_${MODE}>/${P0_RUNTIME_NAME}")
        endif()
    endforeach()
endforeach()
