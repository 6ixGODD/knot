cmake_minimum_required(VERSION 3.20)
project(knot_p0_json_isolation LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
get_filename_component(KNOT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BUNDLED_HEADER "${KNOT_ROOT}/vendor/nlohmann/json.hpp")
set(HOST_HEADER "${HOST_JSON_DIR}/json.hpp")
file(SHA256 "${BUNDLED_HEADER}" BUNDLED_SHA)
file(SHA256 "${HOST_HEADER}" HOST_SHA)
if(NOT BUNDLED_SHA STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63"
   OR NOT HOST_SHA STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
    message(FATAL_ERROR "Pinned JSON fixture hash mismatch")
endif()

# Reject every JSON/NLOHMANN macro defined or tested by the exact pinned source,
# including internal include guards. Generate only into the build directory.
file(READ "${BUNDLED_HEADER}" JSON_SOURCE)
string(REGEX MATCHALL "(JSON_|NLOHMANN_|INCLUDE_NLOHMANN_)[A-Z0-9_]+" JSON_MACROS "${JSON_SOURCE}")
list(APPEND JSON_MACROS nlohmann)
list(REMOVE_DUPLICATES JSON_MACROS)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/reject_json_pollution.hpp" "// Generated from the pinned vendor source.\n")
foreach(MACRO IN LISTS JSON_MACROS)
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/reject_json_pollution.hpp"
        "#ifdef ${MACRO}\n#error P0_JSON_POLLUTION: ${MACRO}\n#endif\n")
endforeach()

function(codec_settings TARGET)
    target_include_directories(${TARGET} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(${TARGET} PRIVATE "P0_BUNDLED_HEADER=\"${BUNDLED_HEADER}\"")
    set_target_properties(${TARGET} PROPERTIES UNITY_BUILD OFF DISABLE_PRECOMPILE_HEADERS ON)
endfunction()

add_library(private_codec STATIC codec.cpp)
codec_settings(private_codec)
# Even with a hostile project-wide PCH/unity preference, codec_settings wins.
target_precompile_headers(private_codec PRIVATE "${HOST_HEADER}")
get_target_property(CODEC_UNITY private_codec UNITY_BUILD)
get_target_property(CODEC_NO_PCH private_codec DISABLE_PRECOMPILE_HEADERS)
if(CODEC_UNITY OR NOT CODEC_NO_PCH)
    message(FATAL_ERROR "Codec PCH/unity isolation disabled")
endif()
add_executable(no_host no_host.cpp)
target_link_libraries(no_host PRIVATE private_codec)
add_test(NAME no_host COMMAND no_host)
foreach(VARIANT IN ITEMS old old_diagnostics same same_diagnostics)
    if(VARIANT MATCHES "^old")
        set(HEADER "${HOST_HEADER}")
        set(MINOR 11)
    else()
        set(HEADER "${BUNDLED_HEADER}")
        set(MINOR 12)
    endif()
    if(VARIANT MATCHES "diagnostics$")
        set(DIAGNOSTICS 1)
    else()
        set(DIAGNOSTICS 0)
    endif()
    foreach(ORDER IN ITEMS 0 1)
        set(TARGET "${VARIANT}_header_first_${ORDER}")
        add_executable(${TARGET} consumer.cpp)
        target_link_libraries(${TARGET} PRIVATE private_codec)
        target_compile_definitions(${TARGET} PRIVATE "P0_HOST_HEADER=\"${HEADER}\""
            P0_HOST_MINOR=${MINOR} P0_HOST_DIAGNOSTICS=${DIAGNOSTICS} P0_HEADER_FIRST=${ORDER})
        add_test(NAME ${TARGET} COMMAND ${TARGET})
    endforeach()
endforeach()

# These targets must fail with P0_JSON_POLLUTION, not just any compiler error.
foreach(NEGATIVE IN ITEMS diagnostics implicit namespace preincluded pch)
    if(NEGATIVE STREQUAL "preincluded")
        add_library(reject_${NEGATIVE} OBJECT EXCLUDE_FROM_ALL preincluded.cpp)
    else()
        add_library(reject_${NEGATIVE} OBJECT EXCLUDE_FROM_ALL codec.cpp)
    endif()
    codec_settings(reject_${NEGATIVE})
    target_compile_definitions(reject_${NEGATIVE} PRIVATE "P0_HOST_HEADER=\"${HOST_HEADER}\"")
endforeach()
target_compile_definitions(reject_diagnostics PRIVATE JSON_DIAGNOSTICS=1)
target_compile_definitions(reject_implicit PRIVATE JSON_USE_IMPLICIT_CONVERSIONS=1)
target_compile_definitions(reject_namespace PRIVATE NLOHMANN_JSON_NAMESPACE_NO_VERSION=1)
set_target_properties(reject_pch PROPERTIES DISABLE_PRECOMPILE_HEADERS OFF)
target_precompile_headers(reject_pch PRIVATE "${HOST_HEADER}")

file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/artifacts-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:private_codec>\n${CMAKE_LINKER}\n${CMAKE_NM}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/toolchain.txt"
    "compiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\ncompiler_path=${CMAKE_CXX_COMPILER}\nsystem=${CMAKE_SYSTEM_NAME} ${CMAKE_SYSTEM_VERSION} ${CMAKE_SYSTEM_PROCESSOR}\ngenerator_platform=${CMAKE_GENERATOR_PLATFORM}\npointer_bytes=${CMAKE_SIZEOF_VOID_P}\nmsvc_runtime=${CMAKE_MSVC_RUNTIME_LIBRARY}\nrelease_flags=${CMAKE_CXX_FLAGS_RELEASE}\n")
