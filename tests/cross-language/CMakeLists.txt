cmake_minimum_required(VERSION 3.23)
project(knot_cross_language LANGUAGES C CXX)
find_package(Knot CONFIG REQUIRED)
find_package(Threads REQUIRED)
foreach(mode IN ITEMS shared static)
    if(mode STREQUAL "shared")
        set(native Knot::Shared)
        set(bridge Knot::CppShared)
    else()
        set(native Knot::Static)
        set(bridge Knot::CppStatic)
    endif()
    add_executable(c_${mode} c_runner.c)
    target_link_libraries(c_${mode} PRIVATE ${native})
    add_executable(cpp_${mode} cpp_runner.cpp)
    target_include_directories(cpp_${mode} PRIVATE "${HOST_JSON_DIR}")
    target_link_libraries(cpp_${mode} PRIVATE ${bridge} Threads::Threads)
    foreach(target IN ITEMS c_${mode} cpp_${mode})
        set_target_properties(${target} PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
            if(target MATCHES "^c_")
                target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
            else()
                target_compile_options(${target} PRIVATE /EHsc)
            endif()
        endif()
    endforeach()
endforeach()
