cmake_minimum_required(VERSION 3.23)
project(knot_binding_components LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(MSVC)
    add_compile_options(/utf-8)
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
add_subdirectory("${ROOT}/bindings/cpp" bridges)
file(READ "${ROOT}/contracts/golden/T12/expected-result.json" MOCK_RESULT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mock_fixtures.hpp" "static const char mock_transition[]=R\"fixture(${MOCK_RESULT})fixture\";\nstatic const char mock_create[]=R\"fixture({\"schemaVersion\":\"1.0\",\"productId\":\"speaker-pro\",\"productVersion\":\"1\",\"modelDigest\":\"sha256:73e5998863973011c590d9883bb0a593f62d77fca5d10480b4f0029153ed1541\",\"diagnostics\":[]})fixture\";\n")
foreach(mode IN ITEMS shared static)
    if(mode STREQUAL shared)
        add_library(knot_test_boundary_${mode} SHARED native/mock.cpp)
        target_compile_definitions(knot_test_boundary_${mode} PRIVATE KNOT_BUILD)
    else()
        add_library(knot_test_boundary_${mode} STATIC native/mock.cpp)
        target_compile_definitions(knot_test_boundary_${mode} PUBLIC KNOT_STATIC)
    endif()
    target_include_directories(knot_test_boundary_${mode} PUBLIC "${ROOT}/contracts/abi" "${CMAKE_CURRENT_SOURCE_DIR}/native" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    foreach(test IN ITEMS codec ownership error allocation)
        add_executable(${test}_${mode} cpp/${test}.cpp)
        target_include_directories(${test}_${mode} PRIVATE "${ROOT}/vendor/nlohmann")
        target_compile_definitions(${test}_${mode} PRIVATE "CONTRACTS=\"${ROOT}/contracts\"")
        target_link_libraries(${test}_${mode} PRIVATE knot_cpp_${mode} knot_test_boundary_${mode})
        add_test(NAME ${test}_${mode} COMMAND ${test}_${mode})
    endforeach()
endforeach()
add_executable(layout native/layout.c)
target_link_libraries(layout PRIVATE knot_test_boundary_shared)
add_test(NAME layout COMMAND layout)
if(NOT HOST_JSON_DIR)
    message(FATAL_ERROR "Supply HOST_JSON_DIR with the pinned nlohmann 3.11.3 fixture")
endif()
file(SHA256 "${HOST_JSON_DIR}/json.hpp" HOST_SHA)
if(NOT HOST_SHA STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
    message(FATAL_ERROR "Host JSON digest mismatch")
endif()
foreach(version IN ITEMS old same)
    if(version STREQUAL old)
        set(host "${HOST_JSON_DIR}/json.hpp")
        set(minor 11)
    else()
        set(host "${ROOT}/vendor/nlohmann/json.hpp")
        set(minor 12)
    endif()
    foreach(diagnostics IN ITEMS 0 1)
        foreach(first IN ITEMS 0 1)
            set(name "include_${version}_${diagnostics}_${first}")
            add_executable(${name} cpp/include_order.cpp)
            target_compile_definitions(${name} PRIVATE "HOST_HEADER=\"${host}\"" HOST_MINOR=${minor}
                HOST_DIAGNOSTICS=${diagnostics} HEADER_FIRST=${first} "CONTRACTS=\"${ROOT}/contracts\"")
            target_link_libraries(${name} PRIVATE knot_cpp_static knot_test_boundary_static)
            add_test(NAME ${name} COMMAND ${name})
        endforeach()
    endforeach()
endforeach()
# Controlled failed builds audit the actual final codec source, not a probe copy.
foreach(kind IN ITEMS diagnostics implicit namespace preincluded pch)
    add_library(reject_${kind} OBJECT EXCLUDE_FROM_ALL "${ROOT}/bindings/cpp/src/codec.cpp")
    target_include_directories(reject_${kind} PRIVATE "${ROOT}/bindings/cpp/include" "${ROOT}/contracts/abi"
        "${ROOT}/vendor/nlohmann" "${CMAKE_CURRENT_BINARY_DIR}/bridges/generated")
    set_target_properties(reject_${kind} PROPERTIES UNITY_BUILD OFF DISABLE_PRECOMPILE_HEADERS ON)
endforeach()
target_compile_definitions(reject_diagnostics PRIVATE JSON_DIAGNOSTICS=1)
target_compile_definitions(reject_implicit PRIVATE JSON_USE_IMPLICIT_CONVERSIONS=1)
target_compile_definitions(reject_namespace PRIVATE NLOHMANN_JSON_NAMESPACE_NO_VERSION=1)
if(MSVC)
    target_compile_options(reject_preincluded PRIVATE "/FI${HOST_JSON_DIR}/json.hpp")
else()
    target_compile_options(reject_preincluded PRIVATE -include "${HOST_JSON_DIR}/json.hpp")
endif()
set_target_properties(reject_pch PROPERTIES DISABLE_PRECOMPILE_HEADERS OFF)
target_precompile_headers(reject_pch PRIVATE "${HOST_JSON_DIR}/json.hpp")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/fixtures/$<CONFIG>.json" CONTENT
    "{\"library\":\"$<TARGET_FILE:knot_test_boundary_shared>\",\"linkLibrary\":\"$<TARGET_LINKER_FILE:knot_test_boundary_shared>\",\"include\":\"${ROOT}/contracts/abi\",\"sharedBridge\":\"$<TARGET_FILE:knot_cpp_shared>\",\"staticBridge\":\"$<TARGET_FILE:knot_cpp_static>\",\"linker\":\"${CMAKE_LINKER}\",\"nm\":\"${CMAKE_NM}\"}\n")
add_test(NAME archive_and_pollution COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_isolation.py"
    --build-dir "${CMAKE_CURRENT_BINARY_DIR}" --config $<CONFIG> --cmake "${CMAKE_COMMAND}")
set_tests_properties(archive_and_pollution PROPERTIES RUN_SERIAL TRUE TIMEOUT 300)
add_library(c_example OBJECT "${ROOT}/examples/c/smoke.c")
target_include_directories(c_example PRIVATE "${ROOT}/contracts/abi")
add_library(cpp_example OBJECT "${ROOT}/examples/cpp/speaker.cpp")
target_include_directories(cpp_example PRIVATE "${ROOT}/contracts/abi" "${ROOT}/bindings/cpp/include")
add_test(NAME cmake_interface COMMAND "${CMAKE_COMMAND}" -S "${CMAKE_CURRENT_SOURCE_DIR}/cmake-interface"
    -B "${CMAKE_CURRENT_BINARY_DIR}/interface" "-DKNOT_SOURCE_ROOT=${ROOT}")
